{{- $src := .Destination -}} {{- $alt := .Text -}} {{- $title := .Title -}} {{- /* 포스트 본문의 첫 번째 이미지는 대부분 LCP(Largest Contentful Paint) 후보라 lazy loading을 걸면 오히려 발견/로딩이 늦어짐. Page.Store로 페이지당 1회만 감지해 첫 이미지는 eager+fetchpriority=high, 이후 이미지는 기존대로 lazy 처리 */ -}} {{- $isFirstImage := not (.Page.Store.Get "renderedFirstImage") -}} {{- if $isFirstImage -}}{{- .Page.Store.Set "renderedFirstImage" true -}}{{- end -}} {{- if strings.HasPrefix $src "./assets/" -}} {{- $resourcePath := strings.TrimPrefix "./" $src -}} {{- $resource := .Page.Resources.GetMatch $resourcePath -}} {{- if and $resource (reflect.IsImageResourceProcessable $resource) -}} {{- $maxWidth := 1200 -}} {{- $targetWidth := $resource.Width -}} {{- if gt $targetWidth $maxWidth -}}{{- $targetWidth = $maxWidth -}}{{- end -}} {{- $webp := $resource.Resize (printf "%dx webp" $targetWidth) -}} {{- $fallback := $resource.Resize (printf "%dx" $targetWidth) -}} {{- /* 클릭 확대. 원본이 표시 크기보다 클 때만 data-zoom-src 로 원본을 가리킨다. 없으면 medium-zoom 이 이미 받아둔 이미지를 그대로 확대하므로 추가 요청이 없다. */ -}} {{- .Page.Store.Set "hasZoomableImage" true -}} {{- $zoomSrc := "" -}} {{- if gt $resource.Width $targetWidth -}}{{- $zoomSrc = $resource.RelPermalink -}}{{- end -}} {{- else if $resource -}} {{- /* SVG 등 래스터 리사이즈가 불가능한 리소스는 원본 그대로 출력 */}} {{- .Page.Store.Set "hasZoomableImage" true -}} {{- else -}} {{- $basePath := .Page.RelPermalink | strings.TrimPrefix (printf "/%s/" .Page.Language.Lang) -}} {{- if not (strings.HasPrefix $basePath "/") }}{{- $basePath = printf "/%s" $basePath }}{{- end -}} {{- $src = printf "%s%s" $basePath (strings.TrimPrefix "./" $src) -}} {{- end -}} {{- else -}} {{- end -}}
