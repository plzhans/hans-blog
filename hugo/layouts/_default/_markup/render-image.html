{{- $src := .Destination -}} {{- $alt := .Text -}} {{- $title := .Title -}} {{- /* 포스트 본문의 첫 번째 이미지는 대부분 LCP(Largest Contentful Paint) 후보라 lazy loading을 걸면 오히려 발견/로딩이 늦어짐. Page.Store로 페이지당 1회만 감지해 첫 이미지는 eager+fetchpriority=high, 이후 이미지는 기존대로 lazy 처리 */ -}} {{- $isFirstImage := not (.Page.Store.Get "renderedFirstImage") -}} {{- if $isFirstImage -}}{{- .Page.Store.Set "renderedFirstImage" true -}}{{- end -}} {{- if strings.HasPrefix $src "./assets/" -}} {{- $resourcePath := strings.TrimPrefix "./" $src -}} {{- $resource := .Page.Resources.GetMatch $resourcePath -}} {{- if $resource -}} {{- $maxWidth := 1200 -}} {{- $targetWidth := $resource.Width -}} {{- if gt $targetWidth $maxWidth -}}{{- $targetWidth = $maxWidth -}}{{- end -}} {{- $webp := $resource.Resize (printf "%dx webp" $targetWidth) -}} {{- $fallback := $resource.Resize (printf "%dx" $targetWidth) -}} {{- else -}} {{- $basePath := .Page.RelPermalink | strings.TrimPrefix (printf "/%s/" .Page.Language.Lang) -}} {{- if not (strings.HasPrefix $basePath "/") }}{{- $basePath = printf "/%s" $basePath }}{{- end -}} {{- $src = printf "%s%s" $basePath (strings.TrimPrefix "./" $src) -}} {{- end -}} {{- else -}} {{- end -}}
